**Query #3 Burner Accounts Detection**
**Concepts covered: INNER JOIN, explicit date range boundary 0 <= (transaction_time - signup_date).days < 30**

    SELECT 
        t.transaction_id,
        t.user_id,
        u.signup_date,
        t.transaction_time,
        DATEDIFF(t.transaction_time, u.signup_date) AS days_since_signup,
        t.amount_inr,
        t.status
    FROM transactions t
    INNER JOIN users u 
        ON t.user_id = u.user_id
    WHERE t.status = 'chargeback'
      AND t.transaction_time >= u.signup_date
      AND t.transaction_time < DATE_ADD(u.signup_date, INTERVAL 30 DAY)
    ORDER BY t.transaction_time DESC;

**Query #5 Velocity Attacks Detection (10-Minute Windows)**
**Concepts covered: Windowing / Self-Join / Time Bucketing**

    SELECT 
        user_id,
        FROM_UNIXTIME(FLOOR(UNIX_TIMESTAMP(transaction_time) / 600) * 600) AS window_start,
        COUNT(transaction_id) AS transaction_count,
        MIN(transaction_time) AS cluster_earliest_time,
        SUM(amount_inr) AS total_amount
    FROM transactions
    GROUP BY 
        user_id, 
        FROM_UNIXTIME(FLOOR(UNIX_TIMESTAMP(transaction_time) / 600) * 600)
    HAVING COUNT(transaction_id) >= 3
    ORDER BY cluster_earliest_time ASC;

**Query #6 Payment Method Risk Assessment**
**Concepts covered: Combined Aggregations, Filtering**

    SELECT 
        payment_method,
        COUNT(*) AS total_txns,
        SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END) AS failed_txns,
        SUM(CASE WHEN status = 'chargeback' THEN 1 ELSE 0 END) AS chargeback_txns,
        ROUND(AVG(risk_score), 2) AS avg_risk_score
    FROM transactions
    GROUP BY payment_method
